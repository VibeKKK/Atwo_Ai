def top_lnk(root, n):
    items = []
    for dirpath, _dirs, files in os.walk(root):
        for f in files:
            if f.endswith(".lnk"):
                items.append((os.path.getsize(os.path.join(dirpath, f)), f))
    items.sort(reverse=True)
    return items[:n]
